package gov.fdic.tip.emailmanager.controller;

import gov.fdic.tip.emailmanager.constant.AppConstants;
import gov.fdic.tip.emailmanager.dto.ContactDto;
import gov.fdic.tip.emailmanager.service.ContactService;
import jakarta.validation.Valid;
import org.springframework.http.HttpStatus;
import org.springframework.http.ResponseEntity;
import org.springframework.security.access.prepost.PreAuthorize;
import org.springframework.security.core.Authentication;
import org.springframework.web.bind.annotation.*;

import java.util.List;

@RestController
@RequestMapping("/api/v1/contacts")
public class ContactController {

    private final ContactService contactService;

    public ContactController(ContactService contactService) {
        this.contactService = contactService;
    }

    // GET all contacts
    @GetMapping
    @PreAuthorize(AppConstants.PERM_CONTACT_VIEW)
    public ResponseEntity<List<ContactDto>> getAllContacts() {
        return ResponseEntity.ok(contactService.getAllContacts());
    }

    // GET filter contacts by registered dynamic attribute and value
    @GetMapping("/search")
    @PreAuthorize(AppConstants.PERM_CONTACT_VIEW)
    public ResponseEntity<List<ContactDto>> searchContacts(
            @RequestParam Long attributeId,
            @RequestParam String value) {
        return ResponseEntity.ok(contactService.searchContactsByAttribute(attributeId, value));
    }

    // POST create contact
    @PostMapping
    @PreAuthorize(AppConstants.PERM_CONTACT_ADD)
    public ResponseEntity<ContactDto> createContact(
            @Valid @RequestBody ContactDto dto,
            Authentication authentication) {
        ContactDto created = contactService.createContact(dto, authentication.getName());
        return ResponseEntity.status(HttpStatus.CREATED).body(created);
    }

    // PUT update contact
    @PutMapping("/{id}")
    @PreAuthorize(AppConstants.PERM_CONTACT_EDIT)
    public ResponseEntity<ContactDto> updateContact(
            @PathVariable Long id,
            @Valid @RequestBody ContactDto dto,
            Authentication authentication) {
        ContactDto updated = contactService.updateContact(id, dto, authentication.getName());
        return ResponseEntity.ok(updated);
    }

    // DELETE contact (hard delete if unused, soft delete if referenced)
    @DeleteMapping("/{id}")
    @PreAuthorize(AppConstants.PERM_CONTACT_DELETE)
    public ResponseEntity<Void> deleteContact(
            @PathVariable Long id,
            Authentication authentication) {
        contactService.deleteContact(id, authentication.getName());
        return ResponseEntity.noContent().build();
    }
}
