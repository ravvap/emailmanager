package gov.fdic.tip.emailmanager.controller;

import gov.fdic.tip.emailmanager.constant.ActorType;
import gov.fdic.tip.emailmanager.dto.*;
import gov.fdic.tip.emailmanager.service.AuditLogService;
import gov.fdic.tip.emailmanager.service.DistributionListService;
import org.junit.jupiter.api.BeforeEach;
import org.junit.jupiter.api.Test;
import org.junit.jupiter.api.extension.ExtendWith;
import org.mockito.InjectMocks;
import org.mockito.Mock;
import org.mockito.junit.jupiter.MockitoExtension;
import org.springframework.data.domain.Pageable;
import org.springframework.http.HttpStatus;
import org.springframework.http.ResponseEntity;
import org.springframework.security.core.Authentication;

import java.util.List;
import java.util.Map;

import static org.junit.jupiter.api.Assertions.assertEquals;
import static org.junit.jupiter.api.Assertions.assertNotNull;
import static org.mockito.ArgumentMatchers.any;
import static org.mockito.ArgumentMatchers.eq;
import static org.mockito.Mockito.*;

@ExtendWith(MockitoExtension.class)
class DistributionListControllerTest {

    @Mock
    private DistributionListService distributionListService;

    @Mock
    private AuditLogService auditLogService;

    @Mock
    private Authentication authentication;

    @InjectMocks
    private DistributionListController distributionListController;

    private static final String USER_EMAIL = "dl.manager@fdic.gov";

    @BeforeEach
    void setUp() {
        lenient().when(authentication.getName()).thenReturn(USER_EMAIL);
    }

    @Test
    void getAllDistributionLists_Success() {
        DistributionListSummaryDto summary = new DistributionListSummaryDto();
        summary.setId(1L);
        summary.setName("Executive Leadership");

        when(distributionListService.getAllDistributionListsSummary()).thenReturn(List.of(summary));

        ResponseEntity<List<DistributionListSummaryDto>> response = distributionListController.getAllDistributionLists(authentication);

        assertEquals(HttpStatus.OK, response.getStatusCode());
        assertNotNull(response.getBody());
        assertEquals(1, response.getBody().size());

        verify(auditLogService).emitAuditEvent(
                eq("DISTRIBUTION_LIST_READ_ALL"),
                eq("ALL"),
                eq("Distribution Lists Summary List"),
                eq(ActorType.USER),
                eq(USER_EMAIL),
                eq(USER_EMAIL),
                eq(Map.of("count", 1)),
                eq("SUCCESS")
        );
    }

    @Test
    void getDistributionListDetails_Success() {
        Long id = 10L;
        DistributionListViewDto viewDto = new DistributionListViewDto();
        viewDto.setId(id);
        viewDto.setName("IT Operations");

        when(distributionListService.getDistributionListDetails(eq(id), eq("test"), any(Pageable.class))).thenReturn(viewDto);

        ResponseEntity<DistributionListViewDto> response = distributionListController.getDistributionListDetails(id, "test", 0, 10, authentication);

        assertEquals(HttpStatus.OK, response.getStatusCode());
        assertEquals("IT Operations", response.getBody().getName());

        verify(auditLogService).emitAuditEvent(
                eq("DISTRIBUTION_LIST_READ_DETAILS"),
                eq("10"),
                eq("IT Operations"),
                eq(ActorType.USER),
                eq(USER_EMAIL),
                eq(USER_EMAIL),
                eq(Map.of("listId", 10L, "filter", "test", "page", 0, "size", 10)),
                eq("SUCCESS")
        );
    }

    @Test
    void createDistributionList_Success() {
        CreateDistributionListRequest request = new CreateDistributionListRequest();
        request.setName("Security Team");

        DistributionListSummaryDto created = new DistributionListSummaryDto();
        created.setId(20L);
        created.setName("Security Team");
        created.setMemberCount(5L);

        when(distributionListService.createDistributionList(any(CreateDistributionListRequest.class), eq(USER_EMAIL))).thenReturn(created);

        ResponseEntity<DistributionListSummaryDto> response = distributionListController.createDistributionList(request, authentication);

        assertEquals(HttpStatus.CREATED, response.getStatusCode());
        assertEquals(20L, response.getBody().getId());

        verify(auditLogService).emitAuditEvent(
                eq("DISTRIBUTION_LIST_CREATE"),
                eq("20"),
                eq("Security Team"),
                eq(ActorType.USER),
                eq(USER_EMAIL),
                eq(USER_EMAIL),
                eq(Map.of("listName", "Security Team", "memberCount", 5L)),
                eq("SUCCESS")
        );
    }

    @Test
    void deleteDistributionList_Success() {
        Long id = 15L;

        doNothing().when(distributionListService).deleteDistributionList(id, USER_EMAIL);

        ResponseEntity<Void> response = distributionListController.deleteDistributionList(id, authentication);

        assertEquals(HttpStatus.NO_CONTENT, response.getStatusCode());

        verify(auditLogService).emitAuditEvent(
                eq("DISTRIBUTION_LIST_DELETE"),
                eq("15"),
                eq("Distribution List 15"),
                eq(ActorType.USER),
                eq(USER_EMAIL),
                eq(USER_EMAIL),
                eq(Map.of("listId", 15L)),
                eq("SUCCESS")
        );
    }
}
