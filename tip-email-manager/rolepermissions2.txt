INSERT INTO txn.role_permission (role_id, permission_id, version, created_at, created_by)
SELECT r.id, p.id, 1, NOW(), 'system'
FROM txn.app_role r
CROSS JOIN txn.permission p
WHERE 
-- 1. TIP System Administrator (Full Admin Access)
(r.code = 'AZTIPSystemAdminD' AND p.code IN (
    'EM_NOREPLY_DELETE', 'EM_DOMAIN_ADD', 'EM_DOMAIN_EDIT', 
    'EM_APPROVED_SENDER_ADD', 'EM_APPROVED_SENDER_EDIT', 'EM_CONTACT_ATTR_ADD', 
    'EM_CONTACT_ADD', 'EM_CONTACT_EDIT', 'EM_CONTACT_DELETE', 
    'EM_DISTRIBUTION_LIST_VIEW', 'EM_DISTRIBUTION_LIST_ADD', 'EM_DISTRIBUTION_LIST_EDIT', 'EM_DISTRIBUTION_LIST_DELETE', 
    'EM_TEMPLATE_ADD', 'EM_TEMPLATE_EDIT', 'EM_TEMPLATE_DEACTIVATE', 'EM_TEMPLATE_REACTIVATE', 'EM_TEMPLATE_APPROVE', 
    'EM_TEMPLATE_SEND_APPROVED', 'EM_SEND_EXTERNAL_APPROVE', 'EM_SEND_STATUS_TRACK', 
    'EM_TEMPLATE_HISTORY_VIEW', 'EM_TEMPLATE_RESTORE', 'EM_AUDIT_RECORD_VIEW', 'EM_NOREPLY_ADD', 'EM_NOREPLY_EDIT', 'EM_NOREPLY_VIEW',
    'EM_DATA_CONNECTION_ADD', 'EM_DATA_CONNECTION_EDIT', 'EM_DATA_CONNECTION_DELETE', 'EM_DATA_CONNECTION_VIEW', 'EM_DATA_CONNECTION_TEST'
))

OR

-- 2. Compliance Analyst / Sr. Analyst
(r.code = 'TIPComplianceAnalystD' AND p.code IN (
    'EM_CONTACT_ADD', 'EM_CONTACT_EDIT', 'EM_CONTACT_DELETE', 
    'EM_DISTRIBUTION_LIST_VIEW', 'EM_DISTRIBUTION_LIST_ADD', 'EM_DISTRIBUTION_LIST_EDIT', 'EM_DISTRIBUTION_LIST_DELETE', 
    'EM_TEMPLATE_ADD', 'EM_TEMPLATE_EDIT', 'EM_TEMPLATE_DEACTIVATE', 'EM_TEMPLATE_REACTIVATE', 'EM_TEMPLATE_APPROVE', 
    'EM_TEMPLATE_SEND_APPROVED', 'EM_SEND_EXTERNAL_APPROVE', 'EM_SEND_STATUS_TRACK', 
    'EM_TEMPLATE_HISTORY_VIEW', 'EM_TEMPLATE_RESTORE', 
    'EM_QUERY_ADD', 'EM_QUERY_EDIT', 'EM_QUERY_DELETE', 'EM_NOREPLY_VIEW',
    'EM_DATA_CONNECTION_ADD', 'EM_DATA_CONNECTION_EDIT', 'EM_DATA_CONNECTION_VIEW', 'EM_DATA_CONNECTION_TEST'
))

OR

-- 3. Manager
(r.code = 'TIPManagerD' AND p.code IN (
    'EM_CONTACT_ADD', 'EM_CONTACT_EDIT', 'EM_CONTACT_DELETE', 
    'EM_DISTRIBUTION_LIST_VIEW', 'EM_DISTRIBUTION_LIST_ADD', 'EM_DISTRIBUTION_LIST_EDIT', 'EM_DISTRIBUTION_LIST_DELETE', 
    'EM_TEMPLATE_ADD', 'EM_TEMPLATE_EDIT', 'EM_TEMPLATE_DEACTIVATE', 'EM_TEMPLATE_REACTIVATE', 'EM_TEMPLATE_APPROVE', 
    'EM_TEMPLATE_HISTORY_VIEW', 'EM_TEMPLATE_RESTORE','EM_NOREPLY_VIEW',
    'EM_DATA_CONNECTION_VIEW', 'EM_DATA_CONNECTION_TEST'
))

ON CONFLICT (role_id, permission_id) DO NOTHING;
